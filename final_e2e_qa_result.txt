function rememberOrder(orderId){
    if(!orderId) return;

    customerOrders = [
        orderId,
        ...customerOrders.filter(x => x !== orderId)
    ].slice(0, 20);

    saveCustomerOrders();

async function copyMarketText(value){
    const text = String(value || "");

    if(!text){
        return;
    }

    try{
        await navigator.clipboard.writeText(text);
        alert("Copied.");
    }catch(e){
        alert(text);
    }
}

function money(x){
    return "₹" + Number(x || 0).toFixed(0);
}

function escapeHtml(s){
    return String(s || "")
        .replaceAll("&","&amp;")
        .replaceAll("<","&lt;")
        .replaceAll(">","&gt;")
        .replaceAll('"',"&quot;")
        .replaceAll("'","&#039;");
}

async function loadSellers(){

    const box = document.getElementById("sellers");

    try{

        const r = await fetch("/api/market/sellers");

function setCategory(c,button){

    category = c;

    document
        .querySelectorAll(".category")
        .forEach(x => x.classList.remove("active"));

    if(button){
        button.classList.add("active");
    }

    filterProducts();
}

function filterProducts(){

    const q = String(
        document.getElementById("search").value || ""
    ).trim().toLowerCase();

    let list = products.slice();

    if(selectedSeller){
        list = list.filter(
            p => String(p.seller_id || "") === selectedSeller
        );
    }

    if(category !== "all"){

        list = list.filter(p => {

            const text = [
                p.name,
                p.description,
                p.sku
            ]
            .join(" ")
            .toLowerCase();

            return text.includes(category.toLowerCase());
        });
    }

    if(q){

        list = list.filter(p => {

            const text = [
                p.name,
                p.description,
                p.sku,
                p.seller_name,
                p.seller_city
            ]
            .join(" ")
            .toLowerCase();

            return text.includes(q);
        });
    }

    renderProducts(list);
}

function addToCart(id){
    const product = products.find(
        p =>
            String(p.product_id) ===
            String(id)
    );

    if(!product){
        alert("Product not found.");
        return;
    }

    const stock =
        Number(product.stock || 0);

    if(stock <= 0){
        alert(
            "This product is currently out of stock."
        );
        return;
    }

    const existing = cart.find(
        x =>
            String(x.product_id) ===
            String(id)
    );

    if(existing){
        if(
            Number(existing.quantity || 0) >=
            stock
        ){
            alert(
                "Maximum available stock reached."
            );
            return;
        }

        existing.quantity =
            Number(existing.quantity || 0) + 1;

    }else{
        cart.push({
            product_id: product.product_id,
            product_name: product.name,
            seller_id: product.seller_id,
            seller_name: product.seller_name,
            unit_price: Number(product.price || 0),
            quantity: 1,
            stock: stock
        });
    }

    saveCart();
    updateCartCount();
    renderCart();
}


function saveCart(){
    localStorage.setItem(
        "delyvo_cart",
        JSON.stringify(cart)
    );
}

function loadCart(){

    try{
        cart = JSON.parse(
            localStorage.getItem("delyvo_cart") || "[]"
        );
    }catch{
        cart = [];
    }

    updateCartCount();
}

function removeUnavailableCartItems(){

    const before = cart.length;

    cart = cart.filter(item => {

        const stock = Number(item.stock || 0);

        return stock > 0 &&
               Number(item.quantity || 0) > 0;
    });

    if(before !== cart.length){
        saveCart();
        updateCartCount();
    }
}


function updateCartProductsStock(){
    cart = cart.map(item => {
        const p = products.find(
            x =>
                String(x.product_id) ===
                String(item.product_id)
        );

        if(p){
            item.product_name = p.name;
            item.seller_id = p.seller_id;
            item.seller_name = p.seller_name;
            item.unit_price = Number(p.price || 0);
            item.stock = Number(p.stock || 0);

            if(item.quantity > item.stock){
                item.quantity = item.stock;
            }
        }

        return item;
    }).filter(
        item => Number(item.stock || 0) > 0 &&
                Number(item.quantity || 0) > 0
    );

    saveCart();
    updateCartCount();
}


function updateCartCount(){

    document.getElementById("cartCount")
        .textContent =
        cart.reduce(
            (a,b) => a + Number(b.quantity || 0),
            0
        );
}

function openCart(){

    renderCart();

    document.getElementById("cartModal")
        .style.display = "flex";
}

function renderCart(){

    const box =
        document.getElementById("cartItems");

    if(!cart.length){

        box.innerHTML =
            `<div class="empty">
                Cart is empty.
             </div>`;

        document.getElementById("cartTotal")
            .innerHTML = "";

        return;
    }

    box.innerHTML = cart.map((x,i) => `

        <div class="item">

            <div>
                <b>${escapeHtml(x.name)}</b><br>
                🏪 ${escapeHtml(x.seller_name || "")}<br>
                ${money(x.price)} × ${x.quantity}
            </div>

            <div class="qty">

                <button onclick="changeQty(${i},-1)">
                    −
                </button>

                ${x.quantity}

                <button onclick="changeQty(${i},1)">
                    +
                </button>

            </div>

        </div>

    `).join("");

    const subtotal =
        cart.reduce(
            (a,b) =>
                a +
                Number(b.price) *
                Number(b.quantity),
            0
        );

    document.getElementById("cartTotal")
        .innerHTML =
        `<h3>Subtotal: ${money(subtotal)}</h3>`;
}


function changeQty(i,d){
    if(!cart[i]) return;

    const next =
        Number(cart[i].quantity || 0) +
        Number(d || 0);

    if(next <= 0){
        cart.splice(i,1);
    }else{
        const stock =
            Number(cart[i].stock || 0);

        if(stock > 0 && next > stock){
            alert(
                "Maximum available stock reached."
            );
            return;
        }

        cart[i].quantity = next;
    }

    saveCart();
    updateCartCount();
    renderCart();
}

function calculateCartTotal(){

    return cart.reduce(
        (sum,item) =>
            sum +
            Number(item.unit_price || 0) *
            Number(item.quantity || 0),
        0
    );
}

async function placeOrder(){

    updateCartProductsStock();
    removeUnavailableCartItems();

    if(!cart.length){
        alert("Cart is empty.");
        return;
    }

    const customerName =
        document.getElementById("customerName").value.trim();

    const customerPhone =
        document.getElementById("customerPhone").value.trim();

    const address =
        document.getElementById("address").value.trim();

    const paymentType =
        document.getElementById("paymentType").value;

    if(!customerName){
        alert("Please enter your name.");
        return;
    }

    if(!/^[0-9]{10}$/.test(customerPhone)){
        alert("Please enter a valid 10-digit phone number.");
        return;
    }

    if(!address){
        alert("Please enter delivery address.");
        return;
    }

    if(!["COD","PREPAID"].includes(paymentType)){
        alert("Please select a valid payment method.");
        return;
    }



    if(!cart.length){

        alert("Cart is empty");
        return;
    }

    const customer_name =
        document
            .getElementById("customerName")
            .value
            .trim();

    const customer_phone =
        document
            .getElementById("customerPhone")
            .value
            .trim();

    const delivery_address =
        document
            .getElementById("address")
            .value
            .trim();

    const payment_type =
        document
            .getElementById("paymentType")
            .value;

    if(
        !customer_name ||
        !customer_phone ||
        !delivery_address
    ){

        alert("Please fill all delivery details");
        return;
    }

    /*
      Current backend seller-order API
      accepts one seller per order.
      Therefore mixed-seller cart is split
      into separate orders.
    */

    const groups = {};

    cart.forEach(x => {

        if(!groups[x.seller_id]){
            groups[x.seller_id] = [];
        }

        groups[x.seller_id].push({
            product_id:x.product_id,
            quantity:x.quantity
        });

    });

    const createdOrders = [];

    try{

        for(const sellerId of Object.keys(groups)){

            const r =
                await fetch(
                    "/api/sellers/" +
                    encodeURIComponent(sellerId) +
                    "/orders",
                    {
                        method:"POST",
                        headers:{
                            "Content-Type":
                                "application/json"
                        },
                        body:JSON.stringify({
                            customer_name,
                            customer_phone,
                            delivery_address,
                            payment_type,
                            delivery_fee:50,
                            items:groups[sellerId]
                        })
                    }
                );

            const data = await r.json();

            if(!r.ok || data.success === false){

                throw new Error(
                    data.message ||
                    "Order failed"
                );
            }

            const order =
                data.order || {};

            createdOrders.push(
                data.order_id ||
                order.order_id ||
                "ORDER"
            );
        }

        cart = [];
        saveCart();
        updateCartCount();

        closeModal("checkoutModal");

        createdOrders.forEach(
            id => rememberOrder(id)
        );

        
createdOrders.forEach(id => {
    rememberOrder(id);
});

alert(
    "Order placed successfully!

" +
    "Order IDs:
" +
    createdOrders.join("
")
);


        closeModal("checkoutModal");
        closeModal("cartModal");

        loadProducts();

    }catch(e){

        alert(
            e.message ||
            "Server connection error"
        );
    }
}

async function syncCustomerOrders(){

    if(!customerOrders.length){
        return;
    }

    try{

        const r = await fetch("/api/market/customer/orders", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({
                order_ids: customerOrders
            })
        });

        const data = await r.json();

        if(!r.ok || data.success === false){
            return;
        }

        const serverIds = (data.orders || [])
            .map(x => x.order_id)
            .filter(Boolean);

        customerOrders = [
            ...serverIds,
            ...customerOrders.filter(x => !serverIds.includes(x))
        ].slice(0, 20);

        saveCustomerOrders();

    }catch(e){
        console.log("Customer order sync unavailable");
    }
}

function marketShipmentProgress(status){
    const value = String(status || "").toUpperCase();

    const steps = [
        "CREATED",
        "ASSIGNED",
        "PICKED_UP",
        "IN_TRANSIT",
        "OUT_FOR_DELIVERY",
        "DELIVERED"
    ];

    const cancelled = [
        "CANCELLED",
        "RETURN_REQUESTED",
        "RETURNED"
    ];

    if(cancelled.includes(value)){
        return {
            current: value,
            steps: []
        };
    }

    const index = steps.indexOf(value);

    return {
        current: value,
        steps: steps.map((step, i) => ({
            status: step,
            done: index >= 0 && i <= index,
            current: i === index
        }))
    };
}


function renderShipmentProgress(status){
    const progress = marketShipmentProgress(status);

    if(!progress.steps.length){
        return `
            <div class="track">
                <b>${escapeHtml(
                    marketShipmentStatusLabel(progress.current)
                )}</b>
            </div>
        `;
    }

    return `
        <div style="margin:12px 0;">
            ${progress.steps.map(step => `
                <div
                    class="track"
                    style="margin:6px 0;${step.current ? "font-weight:700;" : ""}"
                >
                    ${step.done ? "✓" : "○"}
                    ${escapeHtml(
                        marketShipmentStatusLabel(step.status)
                    )}
                </div>
            `).join("")}
        </div>
    `;
}

        const r =
            await fetch(
                "/api/market/orders/" +
                encodeURIComponent(orderId)
            );

        const data = await r.json();

        if(!r.ok || data.success === false){

            throw new Error(
                data.message ||
                "Order not found"
            );
        }

        const items = data.items || [];
